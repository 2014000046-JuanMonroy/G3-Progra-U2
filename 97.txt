using System;

class Program
{
    static void Main()
    {
        Console.Write("Ingresa un número: ");
        int numero = int.Parse(Console.ReadLine());

        Console.WriteLine($"\nLos primeros 20 múltiplos de {numero} son:\n");

        for (int i = 1; i <= 20; i++)
        {
            Console.WriteLine(numero * i);
        }

        Console.ReadKey();
    }
}
