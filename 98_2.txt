using System;

class Program
{
    static void Main()
    {
        Console.Write("Ingrese un número: ");
        int num = int.Parse(Console.ReadLine());

        int suma = 0;

        for (int i = num; i >= 1; i--)
        {
            Console.WriteLine(i);
            suma += i;
        }

        Console.WriteLine("La suma es: " + suma);
        Console.ReadKey();
    }
}
