using System;

class Program
{
    static void Main()
    {
        int opcion;
        int boletos = 0;
        int total = 0;

        do
        {
            Console.WriteLine("Seleccione una opción:");
            Console.WriteLine("1. Niño ($10)");
            Console.WriteLine("2. Adulto ($15)");
            Console.WriteLine("3. Salir");
            Console.Write("Opción: ");
            
            opcion = int.Parse(Console.ReadLine());

            switch (opcion)
            {
                case 1:
                    boletos++;
                    total += 10;
                    break;

                case 2:
                    boletos++;
                    total += 15;
                    break;

                case 3:
                    Console.WriteLine("\nPrograma finalizado.");
                    break;

                default:
                    Console.WriteLine("Opción inválida.");
                    break;
            }

        } while (opcion != 3);

        Console.WriteLine("\nBoletos vendidos: " + boletos);
        Console.WriteLine("Total cobrado: $" + total);

        Console.ReadKey();
    }
}
