using System;

class Program
{
    static void Main()
    {
        int suma = 0;

        for (int num = 2; num <= 50; num++)
        {
            bool esPrimo = true;

            for (int i = 2; i < num; i++)
            {
                if (num % i == 0)
                {
                    esPrimo = false;
                    break;
                }
            }

            if (esPrimo)
            {
                suma += num;
            }
        }

        Console.WriteLine("La suma de los números primos entre 1 y 50 es: " + suma);
        Console.ReadKey();
    }
}
